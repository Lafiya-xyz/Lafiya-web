## Issue #867: Purpose-Bound Data Access with RLS

### Problem Statement

**Current State:**
- Database queries check **who** is accessing data (user authentication)
- No enforcement of **why** data is being accessed (purpose)
- Analytics code can read clinical PHI columns
- Support queries can access more data than necessary
- No audit trail of access purposes

**Privacy Risk:**
```sql
-- Current: Anyone authenticated can read everything
SELECT allergy, diagnosis, medication FROM patient_records WHERE patient_id = $1;

-- Problem: Same query used for:
-- - Emergency display (legitimate)
-- - Analytics dashboard (over-collection)
-- - Support debugging (excessive access)
-- - Payout verification (purpose mismatch)
```

**Compliance Requirements:**
- **GDPR Article 5:** Purpose limitation - data must be collected for specified purposes
- **HIPAA Minimum Necessary:** Only access PHI needed for the specific purpose
- **CCPA:** Purpose must be disclosed and enforced at collection time

**Impact:**
- Critical: GDPR/HIPAA compliance violation
- Security: Reduces blast radius of compromised code paths
- Audit: Clear purpose trail for every PHI access

### Technical Analysis

**Purpose Taxonomy:**
```typescript
enum DataAccessPurpose {
  EMERGENCY_DISPLAY = 'emergency_display',      // Show data during medical emergency
  PATIENT_SELF_SERVICE = 'patient_self_service', // Patient viewing own data
  VERIFICATION = 'verification',                 // Verify identity/credentials
  PAYOUT_PROCESSING = 'payout_processing',       // Calculate provider payments
  SUPPORT_INVESTIGATION = 'support_investigation', // Customer support queries
  ANALYTICS = 'analytics',                       // Aggregate statistics (no PII)
  AUDIT = 'audit',                              // Security audit queries
  SYSTEM_MAINTENANCE = 'system_maintenance'      // Data cleanup, migrations
}
```

**Purpose-Column Matrix:**
```
Purpose                  | PHI Columns Allowed
-------------------------|--------------------
EMERGENCY_DISPLAY        | All clinical fields
PATIENT_SELF_SERVICE     | Own data only
VERIFICATION             | Name, DOB, photo
PAYOUT_PROCESSING        | Visit count, timestamps (no diagnoses)
SUPPORT_INVESTIGATION    | Redacted data only
ANALYTICS                | No PHI (aggregates only)
AUDIT                    | Access logs, not content
SYSTEM_MAINTENANCE       | Minimal (IDs, metadata)
```

**Row-Level Security (RLS) Enforcement:**
PostgreSQL RLS policies will enforce purpose-based access at the database level, making it impossible to bypass even with direct SQL access.

```typescript
/**
 * Data access purposes for GDPR Article 5 compliance.
 * Every database query must specify a purpose.
 */
export enum DataAccessPurpose {
  EMERGENCY_DISPLAY = 'emergency_display',
  PATIENT_SELF_SERVICE = 'patient_self_service',
  VERIFICATION = 'verification',
  PAYOUT_PROCESSING = 'payout_processing',
  SUPPORT_INVESTIGATION = 'support_investigation',
  ANALYTICS = 'analytics',
  AUDIT = 'audit',
  SYSTEM_MAINTENANCE = 'system_maintenance'
}

/**
 * Column access permissions per purpose.
 * Enforced by RLS policies.
 */
export const PURPOSE_COLUMN_ACCESS: Record<
  DataAccessPurpose,
  {
    allowedTables: string[];
    allowedColumns: Record<string, string[]>;
    maxRows?: number;
    requiresAudit: boolean;
  }
> = {
  [DataAccessPurpose.EMERGENCY_DISPLAY]: {
    allowedTables: ['patient_profiles', 'medical_records', 'medications', 'allergies'],
    allowedColumns: {
      patient_profiles: ['*'], // All columns
      medical_records: ['*'],
      medications: ['*'],
      allergies: ['*']
    },
    requiresAudit: true
  },
  [DataAccessPurpose.PATIENT_SELF_SERVICE]: {
    allowedTables: ['patient_profiles', 'medical_records', 'appointments'],
    allowedColumns: {
      patient_profiles: ['*'], // Own data only (enforced by RLS)
      medical_records: ['*'],
      appointments: ['*']
    },
    requiresAudit: false
  },
  [DataAccessPurpose.VERIFICATION]: {
    allowedTables: ['patient_profiles'],
    allowedColumns: {
      patient_profiles: ['id', 'full_name', 'date_of_birth', 'photo_url']
    },
    requiresAudit: true
  },
  [DataAccessPurpose.PAYOUT_PROCESSING]: {
    allowedTables: ['appointments', 'visits'],
    allowedColumns: {
      appointments: ['id', 'provider_id', 'timestamp', 'duration', 'fee'],
      visits: ['id', 'provider_id', 'timestamp']
      // No diagnosis, medications, allergies
    },
    requiresAudit: true
  },
  [DataAccessPurpose.SUPPORT_INVESTIGATION]: {
    allowedTables: ['patient_profiles', 'appointments'],
    allowedColumns: {
      patient_profiles: ['id', 'created_at', 'last_login'],
      appointments: ['id', 'timestamp', 'status']
      // Redacted: no names, no PHI
    },
    maxRows: 100,
    requiresAudit: true
  },
  [DataAccessPurpose.ANALYTICS]: {
    allowedTables: ['analytics_aggregates'], // Pre-aggregated only
    allowedColumns: {
      analytics_aggregates: ['*']
    },
    requiresAudit: false
  },
  [DataAccessPurpose.AUDIT]: {
    allowedTables: ['audit_logs', 'access_logs'],
    allowedColumns: {
      audit_logs: ['*'],
      access_logs: ['*']
    },
    requiresAudit: false
  },
  [DataAccessPurpose.SYSTEM_MAINTENANCE]: {
    allowedTables: ['migrations', 'jobs'],
    allowedColumns: {
      migrations: ['*'],
      jobs: ['id', 'status', 'created_at']
    },
    requiresAudit: true
  }
};

/**
 * Validate that a purpose is allowed for a given operation.
 */
export function validatePurpose(
  purpose: DataAccessPurpose,
  table: string,
  columns: string[]
): { valid: boolean; reason?: string } {
  const access = PURPOSE_COLUMN_ACCESS[purpose];

  if (!access.allowedTables.includes(table)) {
    return {
      valid: false,
      reason: `Purpose ${purpose} not allowed for table ${table}`
    };
  }

  const allowedColumns = access.allowedColumns[table];
  if (!allowedColumns) {
    return { valid: false, reason: `No columns defined for ${table}` };
  }

  if (allowedColumns[0] === '*') {
    return { valid: true }; // All columns allowed
  }

  const deniedColumns = columns.filter(col => !allowedColumns.includes(col));
  if (deniedColumns.length > 0) {
    return {
      valid: false,
      reason: `Purpose ${purpose} denied for columns: ${deniedColumns.join(', ')}`
    };
  }

  return { valid: true };
}
```

#### Phase 2: Supabase Client Wrapper
**File:** `lib/db/purposeClient.ts`
```typescript
import { createClient, SupabaseClient } from '@supabase/supabase-js';
import { DataAccessPurpose, PURPOSE_COLUMN_ACCESS } from '@/lib/types/purpose';

/**
 * Supabase client wrapper that enforces purpose-bound access.
 * Sets app.purpose in Postgres session for RLS enforcement.
 */
export class PurposeBoundClient {
  private client: SupabaseClient;

  constructor(client: SupabaseClient) {
    this.client = client;
  }

  /**
   * Execute a query with a specific purpose.
   * Sets purpose in Postgres session via set_config.
   */
  async withPurpose<T>(
    purpose: DataAccessPurpose,
    callback: (client: SupabaseClient) => Promise<T>
  ): Promise<T> {
    const access = PURPOSE_COLUMN_ACCESS[purpose];

    // Set purpose in Postgres session
    const { error: configError } = await this.client.rpc('set_purpose', {
      purpose_value: purpose
    });

    if (configError) {
      throw new Error(`Failed to set purpose: ${configError.message}`);
    }

    try {
      // Execute query with purpose context
      const startTime = Date.now();
      const result = await callback(this.client);
      const duration = Date.now() - startTime;

      // Audit if required
      if (access.requiresAudit) {
        await this.auditAccess(purpose, duration);
      }

      return result;
    } finally {
      // Clear purpose after query
      await this.client.rpc('clear_purpose');
    }
  }

  /**
   * Convenience methods for common purposes.
   */
  async forEmergency<T>(callback: (client: SupabaseClient) => Promise<T>): Promise<T> {
    return this.withPurpose(DataAccessPurpose.EMERGENCY_DISPLAY, callback);
  }

  async forPatientSelfService<T>(callback: (client: SupabaseClient) => Promise<T>): Promise<T> {
    return this.withPurpose(DataAccessPurpose.PATIENT_SELF_SERVICE, callback);
  }

  async forVerification<T>(callback: (client: SupabaseClient) => Promise<T>): Promise<T> {
    return this.withPurpose(DataAccessPurpose.VERIFICATION, callback);
  }

  async forPayouts<T>(callback: (client: SupabaseClient) => Promise<T>): Promise<T> {
    return this.withPurpose(DataAccessPurpose.PAYOUT_PROCESSING, callback);
  }

  async forSupport<T>(callback: (client: SupabaseClient) => Promise<T>): Promise<T> {
    return this.withPurpose(DataAccessPurpose.SUPPORT_INVESTIGATION, callback);
  }

  async forAnalytics<T>(callback: (client: SupabaseClient) => Promise<T>): Promise<T> {
    return this.withPurpose(DataAccessPurpose.ANALYTICS, callback);
  }

  private async auditAccess(purpose: DataAccessPurpose, duration: number) {
    await this.client.from('access_audit').insert({
      purpose,
      timestamp: new Date().toISOString(),
      duration_ms: duration,
      user_id: (await this.client.auth.getUser()).data.user?.id
    });
  }
}

/**
 * Factory function to create purpose-bound client.
 */
export function createPurposeBoundClient(supabaseClient: SupabaseClient): PurposeBoundClient {
  return new PurposeBoundClient(supabaseClient);
}
```

```sql
-- Purpose enum type
CREATE TYPE data_access_purpose AS ENUM (
  'emergency_display',
  'patient_self_service',
  'verification',
  'payout_processing',
  'support_investigation',
  'analytics',
  'audit',
  'system_maintenance'
);

-- Function to set purpose in session
CREATE OR REPLACE FUNCTION set_purpose(purpose_value text)
RETURNS void AS $$
BEGIN
  PERFORM set_config('app.purpose', purpose_value, true);
END;
$$ LANGUAGE plpgsql SECURITY DEFINER;

-- Function to get current purpose
CREATE OR REPLACE FUNCTION get_current_purpose()
RETURNS text AS $$
BEGIN
  RETURN current_setting('app.purpose', true);
END;
$$ LANGUAGE plpgsql STABLE;

-- Function to clear purpose
CREATE OR REPLACE FUNCTION clear_purpose()
RETURNS void AS $$
BEGIN
  PERFORM set_config('app.purpose', '', true);
END;
$$ LANGUAGE plpgsql SECURITY DEFINER;

-- RLS Policy: patient_profiles
ALTER TABLE patient_profiles ENABLE ROW LEVEL SECURITY;

-- Emergency display: all columns
CREATE POLICY patient_profiles_emergency ON patient_profiles
FOR SELECT
USING (
  get_current_purpose() = 'emergency_display'
  AND auth.uid() IS NOT NULL
);

-- Patient self-service: own data only
CREATE POLICY patient_profiles_self_service ON patient_profiles
FOR SELECT
USING (
  get_current_purpose() = 'patient_self_service'
  AND auth.uid() = user_id
);

-- Verification: limited columns (enforced by SELECT list)
CREATE POLICY patient_profiles_verification ON patient_profiles
FOR SELECT
USING (
  get_current_purpose() = 'verification'
  AND auth.uid() IS NOT NULL
);

-- Support: redacted data only (IDs, timestamps)
CREATE POLICY patient_profiles_support ON patient_profiles
FOR SELECT
USING (
  get_current_purpose() = 'support_investigation'
  AND auth.uid() IN (SELECT user_id FROM support_staff)
);

-- RLS Policy: medical_records (PHI)
ALTER TABLE medical_records ENABLE ROW LEVEL SECURITY;

CREATE POLICY medical_records_emergency ON medical_records
FOR SELECT
USING (
  get_current_purpose() = 'emergency_display'
  AND EXISTS (
    SELECT 1 FROM patient_profiles
    WHERE patient_profiles.id = medical_records.patient_id
    AND patient_profiles.emergency_access_enabled = true
  )
);

CREATE POLICY medical_records_self_service ON medical_records
FOR SELECT
USING (
  get_current_purpose() = 'patient_self_service'
  AND EXISTS (
    SELECT 1 FROM patient_profiles
    WHERE patient_profiles.id = medical_records.patient_id
    AND patient_profiles.user_id = auth.uid()
  )
);

-- DENY medical records for analytics, support, payouts
CREATE POLICY medical_records_deny_analytics ON medical_records
FOR SELECT
USING (
  get_current_purpose() NOT IN ('analytics', 'support_investigation', 'payout_processing')
);

-- RLS Policy: appointments (semi-sensitive)
ALTER TABLE appointments ENABLE ROW LEVEL SECURITY;

CREATE POLICY appointments_payout ON appointments
FOR SELECT
USING (
  get_current_purpose() = 'payout_processing'
  AND auth.uid() IN (SELECT user_id FROM finance_staff)
);

CREATE POLICY appointments_support ON appointments
FOR SELECT
USING (
  get_current_purpose() = 'support_investigation'
  AND auth.uid() IN (SELECT user_id FROM support_staff)
);

-- Audit table for purpose tracking
CREATE TABLE access_audit (
  id BIGSERIAL PRIMARY KEY,
  user_id UUID REFERENCES auth.users(id),
  purpose data_access_purpose NOT NULL,
  timestamp TIMESTAMPTZ NOT NULL DEFAULT NOW(),
  duration_ms INTEGER,
  query_text TEXT,
  row_count INTEGER,
  INDEX idx_purpose (purpose),
  INDEX idx_timestamp (timestamp),
  INDEX idx_user_id (user_id)
);

-- Enable RLS on audit table (admin only)
ALTER TABLE access_audit ENABLE ROW LEVEL SECURITY;

CREATE POLICY access_audit_read ON access_audit
FOR SELECT
USING (
  auth.uid() IN (SELECT user_id FROM admin_users)
  OR auth.uid() = user_id -- Users can see their own audit logs
);
```

#### Phase 5: Middleware for Purpose Validation
**File:** `middleware.ts`
```typescript
import { NextResponse } from 'next/server';
import type { NextRequest } from 'next/server';
import { DataAccessPurpose } from '@/lib/types/purpose';

/**
 * Route-to-purpose mapping.
 * Every route must declare its data access purpose.
 */
const ROUTE_PURPOSES: Record<string, DataAccessPurpose> = {
  '/api/emergency': DataAccessPurpose.EMERGENCY_DISPLAY,
  '/api/profile': DataAccessPurpose.PATIENT_SELF_SERVICE,
  '/api/verify': DataAccessPurpose.VERIFICATION,
  '/api/payouts': DataAccessPurpose.PAYOUT_PROCESSING,
  '/api/support': DataAccessPurpose.SUPPORT_INVESTIGATION,
  '/api/analytics': DataAccessPurpose.ANALYTICS,
  '/api/audit': DataAccessPurpose.AUDIT
};

export function middleware(request: NextRequest) {
  const pathname = request.nextUrl.pathname;

  // Find matching purpose
  const purpose = Object.entries(ROUTE_PURPOSES).find(([route]) =>
    pathname.startsWith(route)
  )?.[1];

  if (!purpose) {
    console.warn(`No purpose defined for route: ${pathname}`);
  }

  // Attach purpose to request headers for downstream use
  const response = NextResponse.next();
  if (purpose) {
    response.headers.set('X-Data-Purpose', purpose);
  }

  return response;
}

export const config = {
  matcher: '/api/:path*'
};
```

## Issue #868: Detect PHI in URLs and Query Strings

### Problem Statement

**Current State:**
- URLs constructed with string concatenation or template literals
- No validation that PHI data doesn't leak into URLs
- Query parameters may contain patient names, allergies, diagnoses
- Redirect targets logged by CDNs, proxies, analytics, browsers
- URL history persisted in browser indefinitely

**Leakage Scenarios:**
```typescript
// VULNERABLE: PHI in query parameters
router.push(`/profile?allergy=${patientAllergy}`);
// URL logged: /profile?allergy=penicillin (PHI!)
```

```javascript
/**
 * ESLint rule: no-phi-in-urls
 * Detects PHI-typed values used in URL construction.
 */
module.exports = {
  meta: {
    type: 'problem',
    docs: {
      description: 'Prevent PHI data from being placed in URLs',
      category: 'Security',
      recommended: true
    },
    messages: {
      phiInUrl: 'PHI-typed value "{{name}}" used in URL construction. PHI must not appear in URLs.',
      piiInUrl: 'PII-typed value "{{name}}" used in URL construction. Consider using opaque identifiers.',
      unsafeFunction: 'Function "{{func}}" may leak data into URLs. Use safe alternatives.'
    },
    schema: []
  },

  create(context) {
    // Track PHI/PII typed variables
    const phiVariables = new Set();
    const piiVariables = new Set();

    return {
      // Track variable declarations with PHI/PII types
      VariableDeclarator(node) {
        if (node.id.typeAnnotation) {
          const typeName = getTypeName(node.id.typeAnnotation);
          if (typeName.includes('PHI<')) {
            phiVariables.add(node.id.name);
          } else if (typeName.includes('PII<')) {
            piiVariables.add(node.id.name);
          }
        }
      },

      // Check redirect() calls
      CallExpression(node) {
        if (isRedirectFunction(node.callee)) {
          checkArgumentsForPHI(node.arguments, phiVariables, piiVariables);
        }

        // Check router.push()
        if (isRouterPush(node.callee)) {
          checkArgumentsForPHI(node.arguments, phiVariables, piiVariables);
        }

        // Check URLSearchParams constructor
        if (isURLSearchParams(node.callee)) {
          checkArgumentsForPHI(node.arguments, phiVariables, piiVariables);
        }

        // Check NextResponse.redirect()
        if (isNextResponseRedirect(node.callee)) {
          checkArgumentsForPHI(node.arguments, phiVariables, piiVariables);
        }
      },

      // Check template literals used for URLs
      TemplateLiteral(node) {
        if (looksLikeUrl(node)) {
          node.expressions.forEach(expr => {
            if (expr.type === 'Identifier' && phiVariables.has(expr.name)) {
              context.report({
                node: expr,
                messageId: 'phiInUrl',
                data: { name: expr.name }
              });
            }
            if (expr.type === 'Identifier' && piiVariables.has(expr.name)) {
              context.report({
                node: expr,
                messageId: 'piiInUrl',
                data: { name: expr.name }
              });
            }
          });
        }
      }
    };

    function isRedirectFunction(callee) {
      return callee.type === 'Identifier' && callee.name === 'redirect';
    }

    function isRouterPush(callee) {
      return (
        callee.type === 'MemberExpression' &&
        callee.property.name === 'push' &&
        (callee.object.name === 'router' || callee.object.name === 'Router')
      );
    }

    function isURLSearchParams(callee) {
      return callee.type === 'Identifier' && callee.name === 'URLSearchParams';
    }

    function isNextResponseRedirect(callee) {
      return (
        callee.type === 'MemberExpression' &&
        callee.object.name === 'NextResponse' &&
        callee.property.name === 'redirect'
      );
    }

    function checkArgumentsForPHI(args, phiVars, piiVars) {
      args.forEach(arg => {
        if (arg.type === 'Identifier') {
          if (phiVars.has(arg.name)) {
            context.report({
              node: arg,
              messageId: 'phiInUrl',
              data: { name: arg.name }
            });
          } else if (piiVars.has(arg.name)) {
            context.report({
              node: arg,
              messageId: 'piiInUrl',
              data: { name: arg.name }
            });
          }
        }

        // Check object properties
        if (arg.type === 'ObjectExpression') {
          arg.properties.forEach(prop => {
            if (prop.value.type === 'Identifier') {
              if (phiVars.has(prop.value.name)) {
                context.report({
                  node: prop.value,
                  messageId: 'phiInUrl',
                  data: { name: prop.value.name }
                });
              }
            }
          });
        }
      });
    }

    function looksLikeUrl(node) {
      const str = node.quasis[0]?.value.raw || '';
      return str.startsWith('/') || str.startsWith('http');
    }

    function getTypeName(typeAnnotation) {
      // Extract type name from TypeScript AST
      return context.getSourceCode().getText(typeAnnotation);
    }
  }
};
```

```typescript
/**
 * Patterns that indicate potential PHI in URLs.
 * Only runs in development/staging.
 */
const PHI_PATTERNS = [
  /allerg(y|ies)=/i,
  /diagnos(is|es)=/i,
  /medication=/i,
  /condition=/i,
  /symptom=/i,
  /prescription=/i,
  /treatment=/i,
  /disease=/i,
  /illness=/i,
  /phone=\d{3}[-.]?\d{3}[-.]?\d{4}/i, // US phone pattern
  /ssn=\d{3}-\d{2}-\d{4}/i,           // SSN pattern
  /dob=\d{4}-\d{2}-\d{2}/i,           // Date of birth
  /name=[A-Z][a-z]+ [A-Z][a-z]+/      // Full names
];

/**
 * Runtime middleware to detect PHI in URLs.
 * Throws errors in development, logs warnings in staging.
 */
export function urlScannerMiddleware(request: NextRequest) {
  if (process.env.NODE_ENV === 'production') {
    return NextResponse.next(); // Disabled in production (performance)
  }

  const url = request.nextUrl;
  const fullUrl = url.toString();
  const searchParams = url.searchParams.toString();

  // Check URL for PHI patterns
  const violations = PHI_PATTERNS.filter(pattern =>
    pattern.test(fullUrl) || pattern.test(searchParams)
  );

  if (violations.length > 0) {
    const error = new Error(
      `🚨 PHI DETECTED IN URL: ${url.pathname}?${searchParams}\n` +
      `Matched patterns: ${violations.map(p => p.source).join(', ')}\n` +
      `This is a HIPAA violation. PHI must not appear in URLs.`
    );

    if (process.env.NODE_ENV === 'development') {
      // Fail fast in development
      console.error(error);
      return new NextResponse(error.message, { status: 500 });
    } else {
      // Log warning in staging
      console.warn(error);
      // Send to monitoring
      reportSecurityViolation({
        type: 'phi_in_url',
        url: fullUrl,
        patterns: violations.map(p => p.source)
      });
    }
  }

  return NextResponse.next();
}

/**
 * Wrapper for redirect() that scans target URL.
 */
export function safeRedirect(url: string): never {
  if (process.env.NODE_ENV !== 'production') {
    const violations = PHI_PATTERNS.filter(pattern => pattern.test(url));
    if (violations.length > 0) {
      throw new Error(
        `PHI detected in redirect target: ${url}\n` +
        `Use POST requests or encrypted state instead.`
      );
    }
  }

  // Use Next.js redirect
  const { redirect } = require('next/navigation');
  redirect(url);
}

/**
 * Wrapper for NextResponse.redirect that scans target.
 */
export function safeResponseRedirect(url: string, status?: number): NextResponse {
  if (process.env.NODE_ENV !== 'production') {
    const violations = PHI_PATTERNS.filter(pattern => pattern.test(url));
    if (violations.length > 0) {
      throw new Error(`PHI detected in redirect: ${url}`);
    }
  }

  return NextResponse.redirect(url, status);
}
```

```typescript
import type { PHI, PII } from '@/lib/types/brand';

/**
 * Type-safe URL builder that rejects PHI/PII.
 * Uses opaque identifiers instead.
 */
export class SafeUrlBuilder {
  private path: string;
  private params: Record<string, string> = {};

  constructor(path: string) {
    this.path = path;
  }

  /**
   * Add query parameter (non-PHI only).
   * PHI/PII types rejected at compile time.
   */
  addParam<T extends string>(
    key: string,
    value: Exclude<T, PHI<any> | PII<any>>
  ): this {
    this.params[key] = value;
    return this;
  }

  /**
   * Add opaque identifier (safe for URLs).
   */
  addId(key: string, id: string | number): this {
    this.params[key] = String(id);
    return this;
  }

  /**
   * Build final URL.
   */
  build(): string {
    const params = new URLSearchParams(this.params);
    const query = params.toString();
    return query ? `${this.path}?${query}` : this.path;
  }

  toString(): string {
    return this.build();
  }
}

/**
 * Factory function for safe URLs.
 */
export function safeUrl(path: string): SafeUrlBuilder {
  return new SafeUrlBuilder(path);
}

 * Example usage:
 * 
 * // ✓ SAFE: Opaque identifiers

 *   .build();
 * 
 * // ✗ COMPILE ERROR: PHI type rejected
```

```typescript
import { encrypt, decrypt } from '@/lib/crypto';

/**
 * Instead of PHI in URLs, use encrypted state tokens.
 * Token passed as opaque identifier, decrypted server-side.
 */
export async function createStateToken<T>(data: T, expiresIn: number = 300): Promise<string> {
  const payload = {
    data,
    exp: Date.now() + expiresIn * 1000
  };

  const encrypted = await encrypt(JSON.stringify(payload));
  return Buffer.from(encrypted).toString('base64url');
}

export async function readStateToken<T>(token: string): Promise<T | null> {
  try {
    const encrypted = Buffer.from(token, 'base64url');
    const decrypted = await decrypt(encrypted);
    const payload = JSON.parse(decrypted);

    // Check expiration
    if (payload.exp < Date.now()) {
      return null; // Expired
    }

    return payload.data as T;
  } catch {
    return null; // Invalid token
  }
}

/**
 * Example usage:
 * 
 * // Server: Create token with PHI
 * const token = await createStateToken({
 *   allergy: patient.allergy,
 *   diagnosis: patient.diagnosis
 * });
 * 
 * // URL contains only opaque token
 * redirect(`/emergency?state=${token}`);
 * 
 * // Server: Decrypt token
 * const data = await readStateToken(params.state);
 * console.log(data.allergy); // PHI retrieved safely
 */
```

## Issue #869: Jitter and Batching for Access Notifications

### Problem Statement

**Current State:**
- Access notifications sent immediately when card is scanned
- Instant notifications reveal precise timing of medical events
- Multiple scans = multiple notifications = reveal event count
- SMS metadata visible to carriers, observers

**Privacy Risk:**
```
Timeline without jitter:
  12:00:00 - Card scanned → Notification sent
  12:00:00 - SMS delivered
  Observer: "Medical emergency at exactly 12:00"

Timeline with multiple scans:
  12:00:00 - Scan 1 → Notification 1
  12:00:30 - Scan 2 → Notification 2
  12:01:00 - Scan 3 → Notification 3
  Observer: "Three scans = serious emergency"
```

**Attack Scenarios:**
1. **Timing Correlation:** Partner sees notification timing, correlates with ER visit
2. **Event Counting:** Multiple notifications reveal severity
3. **Metadata Analysis:** SMS timestamps logged by carriers
4. **Physical Surveillance:** Notifications reveal when patient is incapacitated

**Impact:**
- High: Reveals sensitive medical event timing
- Privacy: GDPR right to privacy in health data
- Safety: Could enable domestic abuse scenarios

### Technical Analysis

**Notification Delivery Policies:**
1. **Immediate:** Safety-critical (emergency contact notified of ER admission)
2. **Batched Digest:** Non-urgent (daily summary of non-emergency scans)
3. **Delayed with Jitter:** Moderate urgency (notify after random 5-30 min delay)

**Jitter Strategy:**
- Random delay between configurable min/max
- Prevents precise timing correlation
- Still delivers notification within reasonable timeframe

**Batching Strategy:**
- Aggregate multiple events into single notification
- Reduce notification count
- Daily/weekly digest for non-urgent access

#### Phase 2: Notification Queue with Leased Jobs

```typescript
import { db } from '@/lib/db';
import { NotificationDeliveryPolicy, AccessNotificationType } from '@/lib/types/notifications';

/**
 * Notification queue entry.
 * Stored in database, processed by leased job runner.
 */
export interface QueuedNotification {
  id: string;
  userId: string;
  type: AccessNotificationType;
  policy: NotificationDeliveryPolicy;
  payload: {
    accessorName: string;
    accessType: string;
    timestamp: string;
    location?: string;
  };
  scheduledFor: Date;  // When to deliver
  createdAt: Date;
  deliveredAt?: Date;
  status: 'pending' | 'delivered' | 'failed';
}

/**
 * Enqueue notification with delivery policy.
 */
export async function enqueueNotification(
  userId: string,
  type: AccessNotificationType,
  payload: QueuedNotification['payload']
): Promise<void> {
  // Get user's notification preferences
  const prefs = await getUserNotificationPreferences(userId);
  const policy = prefs.policies[type];

  if (policy === NotificationDeliveryPolicy.DISABLED) {
    return; // User opted out
  }

  let scheduledFor: Date;

  switch (policy) {
    case NotificationDeliveryPolicy.IMMEDIATE:
      scheduledFor = new Date(); // Deliver now
      break;

    case NotificationDeliveryPolicy.DELAYED_JITTER:
      scheduledFor = addJitter(new Date(), prefs.jitter);
      break;

    case NotificationDeliveryPolicy.BATCH_HOURLY:
      scheduledFor = getNextHourlyTime(prefs.batch.hourlyTime);
      break;

    case NotificationDeliveryPolicy.BATCH_DAILY:
      scheduledFor = getNextDailyTime(prefs.batch.dailyTime);
      break;

    case NotificationDeliveryPolicy.BATCH_WEEKLY:
      scheduledFor = getNextWeeklyTime(prefs.batch.weeklyDay, prefs.batch.weeklyTime);
      break;

    default:
      scheduledFor = new Date();
  }

  // Insert into queue
  await db.notificationQueue.create({
    userId,
    type,
    policy,
    payload,
    scheduledFor,
    status: 'pending'
  });
}

/**
 * Add random jitter to timestamp.
 */
function addJitter(date: Date, jitter: { min: number; max: number }): Date {
  const delayMinutes = Math.random() * (jitter.max - jitter.min) + jitter.min;
  return new Date(date.getTime() + delayMinutes * 60 * 1000);
}

/**
 * Get next hourly delivery time.
 */
function getNextHourlyTime(minuteOfHour: number): Date {
  const now = new Date();
  const next = new Date(now);
  next.setMinutes(minuteOfHour);
  next.setSeconds(0);
  next.setMilliseconds(0);

  // If we passed this hour's time, schedule for next hour
  if (next <= now) {
    next.setHours(next.getHours() + 1);
  }

  return next;
}

/**
 * Get next daily delivery time.
 */
function getNextDailyTime(timeStr: string): Date {
  const [hours, minutes] = timeStr.split(':').map(Number);
  const now = new Date();
  const next = new Date(now);
  next.setHours(hours);
  next.setMinutes(minutes);
  next.setSeconds(0);
  next.setMilliseconds(0);

  // If we passed today's time, schedule for tomorrow
  if (next <= now) {
    next.setDate(next.getDate() + 1);
  }

  return next;
}

/**
 * Get next weekly delivery time.
 */
function getNextWeeklyTime(dayOfWeek: number, timeStr: string): Date {
  const [hours, minutes] = timeStr.split(':').map(Number);
  const now = new Date();
  const next = new Date(now);

  // Calculate days until target day of week
  const daysUntilTarget = (dayOfWeek - now.getDay() + 7) % 7;
  next.setDate(next.getDate() + (daysUntilTarget || 7)); // Next week if today

  next.setHours(hours);
  next.setMinutes(minutes);
  next.setSeconds(0);
  next.setMilliseconds(0);

  return next;
```

```typescript
import { leasedJobRunner } from '@/lib/jobs/leasedRunner';
import { db } from '@/lib/db';
import { sendEmail, sendSMS, sendPush } from '@/lib/notifications/senders';

/**
 * Leased job that processes pending notifications.
 * Runs every minute, processes notifications whose scheduledFor has passed.
 */
export async function processNotificationDigests() {
  const now = new Date();

  // Lease pending notifications
  const notifications = await db.notificationQueue.findMany({
    where: {
      status: 'pending',
      scheduledFor: { lte: now }
    },
    orderBy: { scheduledFor: 'asc' },
    take: 100
  });

  if (notifications.length === 0) {
    return;
  }

  // Group by user and policy for batching
  const grouped = groupNotifications(notifications);

  for (const [userId, groups] of Object.entries(grouped)) {
    const prefs = await getUserNotificationPreferences(userId);

    for (const [policy, items] of Object.entries(groups)) {
      if (shouldBatch(policy)) {
        // Send as digest
        await sendDigestNotification(userId, items, prefs);
      } else {
        // Send individually
        for (const item of items) {
          await sendSingleNotification(userId, item, prefs);
        }
      }

      // Mark as delivered
      await db.notificationQueue.updateMany({
        where: { id: { in: items.map(n => n.id) } },
        data: { status: 'delivered', deliveredAt: new Date() }
      });
    }
  }
}

/**
 * Group notifications by user and policy.
 */
function groupNotifications(
  notifications: QueuedNotification[]
): Record<string, Record<string, QueuedNotification[]>> {
  const result: Record<string, Record<string, QueuedNotification[]>> = {};

  for (const notif of notifications) {
    if (!result[notif.userId]) {
      result[notif.userId] = {};
    }
    if (!result[notif.userId][notif.policy]) {
      result[notif.userId][notif.policy] = [];
    }
    result[notif.userId][notif.policy].push(notif);
  }

  return result;
}

/**
 * Check if policy requires batching.
 */
function shouldBatch(policy: string): boolean {
  return policy.startsWith('batch_');
}

/**
 * Send digest notification (multiple events).
 */
async function sendDigestNotification(
  userId: string,
  notifications: QueuedNotification[],
  prefs: NotificationPreferences
) {
  const user = await db.user.findUnique({ where: { id: userId } });

  const summary = {
    total: notifications.length,
    byType: countByType(notifications),
    period: {
      start: notifications[0].createdAt,
      end: notifications[notifications.length - 1].createdAt
    }
  };

  const message = formatDigestMessage(summary);

  if (prefs.channels.email) {
    await sendEmail({
      to: user.email,
      subject: `Health Data Access Summary - ${notifications.length} events`,
      template: 'access-digest',
      data: { summary, notifications }
    });
  }

  if (prefs.channels.sms) {
    await sendSMS({
      to: user.phone,
      message: `${notifications.length} health data accesses. Check email for details.`
    });
  }

  if (prefs.channels.push) {
    await sendPush({
      userId,
      title: 'Access Summary',
      body: message
    });
  }
}

/**
 * Send single notification.
 */
async function sendSingleNotification(
  userId: string,
  notification: QueuedNotification,
  prefs: NotificationPreferences
) {
  const user = await db.user.findUnique({ where: { id: userId } });
  const message = formatSingleMessage(notification);

  if (prefs.channels.email) {
    await sendEmail({
      to: user.email,
      subject: 'Health Data Access Notification',
      template: 'access-single',
      data: notification.payload
    });
  }

  if (prefs.channels.sms) {
    await sendSMS({ to: user.phone, message });
  }

  if (prefs.channels.push) {
    await sendPush({
      userId,
      title: 'Health Data Accessed',
      body: message
    });
  }
}

function countByType(notifications: QueuedNotification[]): Record<string, number> {
  return notifications.reduce((acc, n) => {
    acc[n.type] = (acc[n.type] || 0) + 1;
    return acc;
  }, {} as Record<string, number>);
}

function formatDigestMessage(summary: any): string {
  return `Your health data was accessed ${summary.total} times in the past period.`;
}

function formatSingleMessage(notification: QueuedNotification): string {
  const { accessorName, accessType, timestamp } = notification.payload;
  return `${accessorName} accessed your ${accessType} data at ${timestamp}.`;
}

// Register job with leased runner (runs every minute)
leasedJobRunner.register('notification-digest', processNotificationDigests, {
  intervalSeconds: 60
});
```

### Testing Strategy

#### Schema Validation Tests
Tests run in CI on every commit:
1. Validate golden files against JSON schemas
2. Generate fresh passes and validate against schemas
3. Compare generated passes to golden files
4. Verify all required fields present

#### Manifest Tests
Tests run in CI with test certificate:
1. Generate manifest SHA-1 hashes
2. Sign manifest with test cert (self-signed)
3. Verify signature with test cert
4. Detect tampering by modifying files

#### Regression Tests
When pass structure changes:
1. Developer runs `npm run generate:golden-files`
2. Reviews diff of golden files
3. Commits new golden files if intentional
4. CI validates new golden files
