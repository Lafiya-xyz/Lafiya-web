## Issue #861: Sandbox PDF/Wallpaper/Pass Generation

### Problem Statement

**Current State:**
- PDF generation, wallpaper creation, and pass generation run in the main Node.js process
- No resource limits or isolation for these operations
- Risk of resource exhaustion attacks through malicious or oversized inputs
- Potential for memory leaks affecting the entire application
- No timeout mechanisms for long-running generation tasks

**Security Concerns:**
1. **Resource Exhaustion:** Malicious users could submit inputs that consume excessive CPU/memory
2. **Process Contamination:** Bugs in generation libraries could crash the main application
3. **Side-Channel Attacks:** Timing attacks through resource consumption patterns
4. **Denial of Service:** Multiple concurrent generation requests could overwhelm the server

**Impact:**
- High: Affects application availability and stability
- Security: Prevents resource-based DoS attacks
- Reliability: Isolates failures in generation tasks

### Technical Analysis

**Current Architecture:**
```javascript
// Current implementation (vulnerable)
async function generatePDF(data) {
  // Runs in main process - no limits
  const pdf = await pdfLibrary.create(data);
  return pdf.toBuffer();
}
```

**Required Changes:**
1. Move generation to isolated worker threads
2. Set CPU and memory limits per worker
3. Implement timeout mechanisms
4. Add resource monitoring and cleanup
5. Handle worker failures gracefully

**Technology Stack:**
- Node.js `worker_threads` module
- `resourceLimits` option for memory/CPU constraints
- Worker pool management for efficiency
- Message passing for safe communication

### Implementation Plan

#### Phase 1: Worker Infrastructure
**File:** `src/workers/sandboxedWorker.js`
```javascript
const { Worker } = require('worker_threads');

class SandboxedWorkerPool {
  constructor(options = {}) {
    this.maxWorkers = options.maxWorkers || 4;
    this.workerTimeout = options.timeout || 30000; // 30s default
    this.resourceLimits = {
      maxOldGenerationSizeMb: options.maxMemoryMb || 256,
      maxYoungGenerationSizeMb: options.maxMemoryMb ? options.maxMemoryMb / 4 : 64,
      codeRangeSizeMb: 16
    };
    this.workers = [];
    this.taskQueue = [];
  }

  async execute(task, data) {
    return new Promise((resolve, reject) => {
      const timeoutId = setTimeout(() => {
        reject(new Error('Worker task timeout'));
      }, this.workerTimeout);

      const worker = new Worker('./workers/generationWorker.js', {
        workerData: { task, data },
        resourceLimits: this.resourceLimits
      });

      worker.on('message', (result) => {
        clearTimeout(timeoutId);
        worker.terminate();
        resolve(result);
      });

      worker.on('error', (error) => {
        clearTimeout(timeoutId);
        worker.terminate();
        reject(error);
      });

      worker.on('exit', (code) => {
        if (code !== 0) {
          clearTimeout(timeoutId);
          reject(new Error(`Worker exited with code ${code}`));
        }
      });
    });
  }
}

module.exports = SandboxedWorkerPool;
```

```javascript
const { parentPort, workerData } = require('worker_threads');
const pdfGenerator = require('../lib/pdfGenerator');
const wallpaperGenerator = require('../lib/wallpaperGenerator');
const passGenerator = require('../lib/passGenerator');

async function executeTask() {
  try {
    const { task, data } = workerData;
    
    let result;
    switch (task) {
      case 'pdf':
        result = await pdfGenerator.generate(data);
        break;
      case 'wallpaper':
        result = await wallpaperGenerator.generate(data);
        break;
      case 'pass':
        result = await passGenerator.generate(data);
        break;
      default:
        throw new Error(`Unknown task type: ${task}`);
    }

    parentPort.postMessage({
      success: true,
      data: result
    });
  } catch (error) {
    parentPort.postMessage({
      success: false,
      error: error.message,
      stack: error.stack
    });
  }
}

executeTask();
```

```javascript
class WorkerMetrics {
  constructor() {
    this.metrics = {
      totalTasks: 0,
      successfulTasks: 0,
      failedTasks: 0,
      timeouts: 0,
      averageExecutionTime: 0,
      peakMemoryUsage: 0
    };
  }

  recordTask(task, duration, success, error) {
    this.metrics.totalTasks++;
    if (success) {
      this.metrics.successfulTasks++;
    } else {
      this.metrics.failedTasks++;
      if (error?.includes('timeout')) {
        this.metrics.timeouts++;
      }
    }

    // Update average execution time
    const prevAvg = this.metrics.averageExecutionTime;
    this.metrics.averageExecutionTime = 
      (prevAvg * (this.metrics.totalTasks - 1) + duration) / this.metrics.totalTasks;
  }

  getMetrics() {
    return {
      ...this.metrics,
      successRate: this.metrics.successfulTasks / this.metrics.totalTasks,
      failureRate: this.metrics.failedTasks / this.metrics.totalTasks,
      timeoutRate: this.metrics.timeouts / this.metrics.totalTasks
    };
  }
}

module.exports = WorkerMetrics;
```

## Issue #862: Detect Anomalous Session Reuse

### Problem Statement

**Current State:**
- Session tokens can be reused from any location/device
- No detection of suspicious session patterns
- Geographic location changes not monitored
- Device fingerprint changes not tracked
- ASN (Autonomous System Number) changes ignored

**Security Concerns:**
1. **Session Hijacking:** Stolen tokens can be used anywhere without detection
2. **Credential Stuffing:** Compromised sessions reused across networks
3. **Account Takeover:** No early warning of suspicious access patterns
4. **Insider Threats:** Legitimate sessions used from unexpected locations

**Attack Scenarios:**
- Token stolen via XSS, used from different country
- Session cookie leaked, accessed from different ISP
- Man-in-the-middle attack from different ASN
- Device compromise detected by fingerprint drift

**Impact:**
- High: Critical for detecting account compromise
- Security: Enables early detection of token theft
- User Protection: Alerts users to suspicious activity

### Technical Analysis

**Detection Signals:**
1. **Geographic Anomaly:** Country/region change between requests
2. **ASN Change:** Different autonomous system (ISP/datacenter)
3. **Device Drift:** Significant changes in device fingerprint
4. **Temporal Anomaly:** Impossible travel time between locations
5. **Behavioral Anomaly:** Usage pattern changes

**Threat Model:**
```
Normal Session:
  Request 1: ASN=12345, Country=US, Device=A, Time=T
  Request 2: ASN=12345, Country=US, Device=A, Time=T+5min
  → No alerts

Hijacked Session:
  Request 1: ASN=12345, Country=US, Device=A, Time=T
  Request 2: ASN=67890, Country=CN, Device=B, Time=T+1min
  → ALERT: Geographic + ASN + Device change
```

```javascript
const geoip = require('geoip-lite');
const UAParser = require('ua-parser-js');
const crypto = require('crypto');

class SessionContext {
  static extractContext(req) {
    const ip = req.ip || req.connection.remoteAddress;
    const geo = geoip.lookup(ip) || {};
    const ua = new UAParser(req.headers['user-agent']);
    
    return {
      ip,
      asn: geo.asn || 'unknown',
      country: geo.country || 'unknown',
      region: geo.region || 'unknown',
      city: geo.city || 'unknown',
      timezone: geo.timezone || 'unknown',
      deviceFingerprint: this.generateFingerprint(req),
      userAgent: req.headers['user-agent'],
      browser: ua.getBrowser().name,
      os: ua.getOS().name,
      device: ua.getDevice().type || 'desktop',
      timestamp: new Date().toISOString()
    };
  }

  static generateFingerprint(req) {
    const components = [
      req.headers['user-agent'],
      req.headers['accept-language'],
      req.headers['accept-encoding'],
      req.headers['accept']
    ].join('|');
    
    return crypto.createHash('sha256')
      .update(components)
      .digest('hex')
      .substring(0, 16);
  }
}

module.exports = SessionContext;
```

#### Phase 2: Anomaly Detection
**File:** `src/security/anomalyDetector.js`
```javascript
class AnomalyDetector {
  constructor() {
    this.riskScores = {
      COUNTRY_CHANGE: 50,
      ASN_CHANGE: 40,
      DEVICE_CHANGE: 30,
      IMPOSSIBLE_TRAVEL: 80,
      TIMEZONE_MISMATCH: 20
    };
    this.threshold = 60; // Alert if score >= 60
  }

  async detectAnomalies(sessionId, currentContext, previousContext) {
    if (!previousContext) {
      return { safe: true, score: 0, anomalies: [] };
    }

    const anomalies = [];
    let totalScore = 0;

    // Check country change
    if (currentContext.country !== previousContext.country) {
      anomalies.push({
        type: 'COUNTRY_CHANGE',
        severity: 'high',
        details: `${previousContext.country} → ${currentContext.country}`,
        score: this.riskScores.COUNTRY_CHANGE
      });
      totalScore += this.riskScores.COUNTRY_CHANGE;
    }

    // Check ASN change (ISP/datacenter change)
    if (currentContext.asn !== previousContext.asn) {
      anomalies.push({
        type: 'ASN_CHANGE',
        severity: 'medium',
        details: `${previousContext.asn} → ${currentContext.asn}`,
        score: this.riskScores.ASN_CHANGE
      });
      totalScore += this.riskScores.ASN_CHANGE;
    }

    // Check device fingerprint change
    if (currentContext.deviceFingerprint !== previousContext.deviceFingerprint) {
      const similarity = this.calculateSimilarity(
        currentContext.deviceFingerprint,
        previousContext.deviceFingerprint
      );
      
      if (similarity < 0.8) { // <80% similar
        anomalies.push({
          type: 'DEVICE_CHANGE',
          severity: 'medium',
          details: `Fingerprint similarity: ${(similarity * 100).toFixed(0)}%`,
          score: this.riskScores.DEVICE_CHANGE
        });
        totalScore += this.riskScores.DEVICE_CHANGE;
      }
    }

    // Check impossible travel
    const travelCheck = this.checkImpossibleTravel(
      previousContext,
      currentContext
    );
    if (travelCheck.impossible) {
      anomalies.push({
        type: 'IMPOSSIBLE_TRAVEL',
        severity: 'critical',
        details: travelCheck.reason,
        score: this.riskScores.IMPOSSIBLE_TRAVEL
      });
      totalScore += this.riskScores.IMPOSSIBLE_TRAVEL;
    }

    // Check timezone mismatch
    if (currentContext.timezone !== previousContext.timezone) {
      anomalies.push({
        type: 'TIMEZONE_MISMATCH',
        severity: 'low',
        details: `${previousContext.timezone} → ${currentContext.timezone}`,
        score: this.riskScores.TIMEZONE_MISMATCH
      });
      totalScore += this.riskScores.TIMEZONE_MISMATCH;
    }

    return {
      safe: totalScore < this.threshold,
      score: totalScore,
      threshold: this.threshold,
      anomalies,
      action: this.determineAction(totalScore)
    };
  }

  calculateSimilarity(fp1, fp2) {
    // Hamming distance for hex fingerprints
    let matches = 0;
    const len = Math.min(fp1.length, fp2.length);
    for (let i = 0; i < len; i++) {
      if (fp1[i] === fp2[i]) matches++;
    }
    return matches / len;
  }

  checkImpossibleTravel(prev, current) {
    const timeDiff = new Date(current.timestamp) - new Date(prev.timestamp);
    const hours = timeDiff / (1000 * 60 * 60);

    // If location changed countries in <1 hour, likely impossible
    if (prev.country !== current.country && hours < 1) {
      return {
        impossible: true,
        reason: `Country change in ${hours.toFixed(1)} hours`
      };
    }

    return { impossible: false };
  }

  determineAction(score) {
    if (score >= 100) return 'TERMINATE_SESSION';
    if (score >= 60) return 'REQUIRE_MFA';
    if (score >= 40) return 'ALERT_USER';
    return 'MONITOR';
  }
}

module.exports = AnomalyDetector;
```

#### Phase 3: Session Middleware
**File:** `src/middleware/sessionSecurity.js`
```javascript
const SessionContext = require('./sessionContext');
const AnomalyDetector = require('../security/anomalyDetector');
const SessionStore = require('../db/sessionStore');

const detector = new AnomalyDetector();

async function sessionSecurityMiddleware(req, res, next) {
  if (!req.session || !req.session.id) {
    return next(); // No session to check
  }

  const currentContext = SessionContext.extractContext(req);
  const previousContext = await SessionStore.getLastContext(req.session.id);

  // Store current context
  await SessionStore.saveContext(req.session.id, currentContext);

  // Detect anomalies
  const analysis = await detector.detectAnomalies(
    req.session.id,
    currentContext,
    previousContext
  );

  if (!analysis.safe) {
    // Log security event
    await SecurityLog.create({
      sessionId: req.session.id,
      userId: req.session.userId,
      anomalies: analysis.anomalies,
      score: analysis.score,
      action: analysis.action,
      context: currentContext
    });

    // Take action based on severity
    switch (analysis.action) {
      case 'TERMINATE_SESSION':
        req.session.destroy();
        return res.status(401).json({
          error: 'Session terminated due to security concerns',
          reason: 'anomalous_activity'
        });

      case 'REQUIRE_MFA':
        req.session.requireMFA = true;
        return res.status(403).json({
          error: 'Additional authentication required',
          reason: 'suspicious_activity',
          anomalies: analysis.anomalies
        });

      case 'ALERT_USER':
        // Send alert email/notification (async)
        alertUser(req.session.userId, analysis);
        break;
    }
  }

  // Attach analysis to request for logging
  req.sessionAnalysis = analysis;
  next();
}

async function alertUser(userId, analysis) {
  // Implementation: Send email/push notification
  const user = await User.findById(userId);
  await EmailService.send({
    to: user.email,
    subject: 'Unusual activity detected on your account',
    template: 'security-alert',
    data: {
      anomalies: analysis.anomalies,
      timestamp: new Date().toISOString()
    }
  });
}

module.exports = sessionSecurityMiddleware;
```

```javascript
const { KMSClient, SignCommand, GetPublicKeyCommand } = require('@aws-sdk/client-kms');

class KMSSigningClient {
  constructor(config = {}) {
    this.client = new KMSClient({
      region: config.region || process.env.AWS_REGION || 'us-east-1',
      credentials: config.credentials
    });
    this.keyId = config.keyId || process.env.KMS_SIGNING_KEY_ID;
    this.signingAlgorithm = config.algorithm || 'RSASSA_PKCS1_V1_5_SHA_256';
  }

  async sign(data) {
    // Convert data to Buffer if string
    const message = Buffer.isBuffer(data) ? data : Buffer.from(data, 'utf8');

    try {
      const command = new SignCommand({
        KeyId: this.keyId,
        Message: message,
        SigningAlgorithm: this.signingAlgorithm
      });

      const response = await this.client.send(command);
      
      return {
        signature: Buffer.from(response.Signature).toString('base64'),
        algorithm: this.signingAlgorithm,
        keyId: this.keyId
      };
    } catch (error) {
      throw new KMSError(`KMS signing failed: ${error.message}`, error);
    }
  }

  async getPublicKey() {
    try {
      const command = new GetPublicKeyCommand({
        KeyId: this.keyId
      });

      const response = await this.client.send(command);
      
      return {
        publicKey: Buffer.from(response.PublicKey).toString('base64'),
        keyUsage: response.KeyUsage,
        signingAlgorithms: response.SigningAlgorithms
      };
    } catch (error) {
      throw new KMSError(`Failed to get public key: ${error.message}`, error);
    }
  }

  async verify(data, signature) {
    // For verification, we use the public key
    const { publicKey } = await this.getPublicKey();
    const crypto = require('crypto');
    
    const verifier = crypto.createVerify('RSA-SHA256');
    verifier.update(data);
    
    return verifier.verify(
      {
        key: Buffer.from(publicKey, 'base64'),
        format: 'der',
        type: 'spki'
      },
      Buffer.from(signature, 'base64')
    );
  }
}

class KMSError extends Error {
  constructor(message, originalError) {
    super(message);
    this.name = 'KMSError';
    this.originalError = originalError;
  }
}

module.exports = { KMSSigningClient, KMSError };
```

```javascript
const { KMSSigningClient } = require('../kms/kmsClient');

class JWTKMSSigner {
  constructor(kmsClient) {
    this.kms = kmsClient || new KMSSigningClient();
  }

  async sign(payload, options = {}) {
    const header = {
      alg: 'RS256',
      typ: 'JWT',
      kid: this.kms.keyId
    };

    const now = Math.floor(Date.now() / 1000);
    const claims = {
      ...payload,
      iat: now,
      exp: options.expiresIn ? now + options.expiresIn : now + 3600,
      iss: options.issuer || 'lafiya-web'
    };

    // Encode header and payload
    const encodedHeader = this.base64UrlEncode(JSON.stringify(header));
    const encodedPayload = this.base64UrlEncode(JSON.stringify(claims));
    
    const signingInput = `${encodedHeader}.${encodedPayload}`;
    
    // Sign with KMS
    const { signature } = await this.kms.sign(signingInput);
    const encodedSignature = this.base64UrlEncode(
      Buffer.from(signature, 'base64')
    );

    return `${signingInput}.${encodedSignature}`;
  }

  async verify(token) {
    const parts = token.split('.');
    if (parts.length !== 3) {
      throw new Error('Invalid JWT format');
    }

    const [encodedHeader, encodedPayload, encodedSignature] = parts;
    const signingInput = `${encodedHeader}.${encodedPayload}`;
    
    // Decode signature from base64url
    const signature = Buffer.from(
      encodedSignature.replace(/-/g, '+').replace(/_/g, '/'),
      'base64'
    ).toString('base64');

    // Verify with KMS public key
    const valid = await this.kms.verify(signingInput, signature);
    
    if (!valid) {
      throw new Error('Invalid JWT signature');
    }

    // Decode and validate payload
    const payload = JSON.parse(
      Buffer.from(encodedPayload, 'base64url').toString('utf8')
    );

    const now = Math.floor(Date.now() / 1000);
    if (payload.exp && payload.exp < now) {
      throw new Error('JWT expired');
    }

    return payload;
  }

  base64UrlEncode(data) {
    return Buffer.from(data)
      .toString('base64')
      .replace(/\+/g, '-')
      .replace(/\//g, '_')
      .replace(/=/g, '');
  }
}

module.exports = JWTKMSSigner;
```

```javascript
const { KMSSigningClient } = require('../kms/kmsClient');
const crypto = require('crypto');

class DocumentSigner {
  constructor(kmsClient) {
    this.kms = kmsClient || new KMSSigningClient();
  }

  async signDocument(document, metadata = {}) {
    // Create canonical representation
    const canonical = this.canonicalize(document);
    
    // Hash the document
    const hash = crypto.createHash('sha256')
      .update(canonical)
      .digest();

    // Sign the hash with KMS
    const { signature, keyId } = await this.kms.sign(hash);

    return {
      document,
      signature,
      metadata: {
        ...metadata,
        signedAt: new Date().toISOString(),
        keyId,
        algorithm: 'RS256-SHA256',
        documentHash: hash.toString('hex')
      }
    };
  }

  async verifyDocument(signedDocument) {
    const { document, signature, metadata } = signedDocument;
    
    // Recompute document hash
    const canonical = this.canonicalize(document);
    const hash = crypto.createHash('sha256')
      .update(canonical)
      .digest();

    // Verify hash matches
    if (hash.toString('hex') !== metadata.documentHash) {
      return {
        valid: false,
        reason: 'Document has been modified'
      };
    }

    // Verify signature with KMS
    const valid = await this.kms.verify(hash, signature);

    return {
      valid,
      reason: valid ? 'Signature valid' : 'Invalid signature',
      metadata
    };
  }

  canonicalize(document) {
    // Sort object keys for consistent hashing
    if (typeof document === 'object') {
      return JSON.stringify(document, Object.keys(document).sort());
    }
    return document.toString();
  }
}

module.exports = DocumentSigner;
```

## Issue #864: Evaluate Device Bound Session Credentials (DBSC)

### Problem Statement

**Current State:**
- Session tokens are bearer tokens (anyone with token has access)
- Tokens can be used from any device if stolen
- No cryptographic binding between session and device
- Session hijacking requires only token theft

**Device Bound Session Credentials (DBSC):**
- Emerging standard for binding sessions to specific devices
- Uses device-generated key pairs
- Session token only valid when presented with device signature
- Prevents token theft/replay attacks

**Evaluation Goals:**
1. Understand DBSC specification and browser support
2. Assess feasibility for Lafiya Web
3. Prototype implementation approach
4. Measure user experience impact
5. Evaluate security benefits vs. complexity

**Impact:**
- High: Significantly reduces session hijacking risk
- Complexity: New cryptographic protocol
- Browser Support: Limited (Chrome Origin Trial as of 2024)

### Technical Analysis

**DBSC Overview:**
```
Traditional Session:
  Client → Server: Cookie: session=abc123
  Server: Validates token, grants access
  Problem: Stolen token = full access

DBSC Session:
  Device generates keypair (pub, priv)
  Client → Server: Cookie: session=abc123, Signature: sign(request, priv)
  Server: Validates token + signature using stored pub key
  Problem solved: Stolen token useless without device private key
```

**Browser Support (2024):**
- Chrome: Origin Trial available
- Firefox: No implementation yet
- Safari: No implementation yet
- Edge: Follows Chrome (Chromium-based)

**Specification Status:**
- IETF Draft: https://datatracker.ietf.org/doc/draft-ietf-oauth-dpop/
- W3C Discussion: https://github.com/WICG/dbsc

```javascript
// Proof-of-concept DBSC implementation

class DBSCClient {
  constructor() {
    this.keyPair = null;
    this.supported = this.checkSupport();
  }

  checkSupport() {
    // Check for DBSC API availability
    return 'credentials' in navigator && 
           'createDBSC' in navigator.credentials;
  }

  async initialize() {
    if (!this.supported) {
      console.warn('DBSC not supported in this browser');
      return false;
    }

    try {
      // Generate device-bound key pair
      this.keyPair = await navigator.credentials.createDBSC({
        challenge: await this.getChallenge(),
        rp: {
          name: 'Lafiya Web',
          id: window.location.hostname
        }
      });

      return true;
    } catch (error) {
      console.error('DBSC initialization failed:', error);
      return false;
    }
  }

  async getChallenge() {
    const response = await fetch('/api/auth/dbsc/challenge');
    const { challenge } = await response.json();
    return challenge;
  }

  async signRequest(url, method, body) {
    if (!this.keyPair) {
      throw new Error('DBSC not initialized');
    }

    // Create signature over request
    const message = `${method} ${url}\n${body || ''}`;
    const encoder = new TextEncoder();
    const data = encoder.encode(message);

    const signature = await crypto.subtle.sign(
      {
        name: 'ECDSA',
        hash: { name: 'SHA-256' }
      },
      this.keyPair.privateKey,
      data
    );

    return {
      signature: btoa(String.fromCharCode(...new Uint8Array(signature))),
      keyId: this.keyPair.id,
      algorithm: 'ES256'
    };
  }

  async fetch(url, options = {}) {
    if (!this.supported) {
      // Fall back to regular fetch
      return fetch(url, options);
    }

    // Sign the request
    const { signature, keyId, algorithm } = await this.signRequest(
      url,
      options.method || 'GET',
      options.body
    );

    // Add DBSC headers
    const headers = {
      ...options.headers,
      'X-Device-Signature': signature,
      'X-Device-Key-Id': keyId,
      'X-Device-Algorithm': algorithm
    };

    return fetch(url, { ...options, headers });
  }
}
```

```javascript
const crypto = require('crypto');

class DBSCVerifier {
  constructor() {
    this.publicKeys = new Map(); // Store device public keys
  }

  async verifyRequest(req) {
    const signature = req.headers['x-device-signature'];
    const keyId = req.headers['x-device-key-id'];
    const algorithm = req.headers['x-device-algorithm'];

    if (!signature || !keyId) {
      // DBSC not used, fall back to regular session validation
      return { verified: false, reason: 'no-dbsc' };
    }

    // Retrieve device public key
    const publicKey = await this.getPublicKey(req.session.userId, keyId);
    if (!publicKey) {
      return { verified: false, reason: 'unknown-device' };
    }

    // Reconstruct signed message
    const message = `${req.method} ${req.originalUrl}\n${req.body || ''}`;

    // Verify signature
    const verifier = crypto.createVerify('SHA256');
    verifier.update(message);
    
    const valid = verifier.verify(
      publicKey,
      Buffer.from(signature, 'base64')
    );

    return {
      verified: valid,
      keyId,
      reason: valid ? 'valid' : 'invalid-signature'
    };
  }

  async getPublicKey(userId, keyId) {
    // Retrieve from database
    const device = await DeviceKey.findOne({
      userId,
      keyId,
      active: true
    });

    return device ? device.publicKey : null;
  }

  async registerDevice(userId, publicKey, deviceInfo) {
    const keyId = crypto.randomBytes(16).toString('hex');
    
    await DeviceKey.create({
      userId,
      keyId,
      publicKey,
      deviceInfo,
      createdAt: new Date(),
      lastUsed: new Date()
    });

    return keyId;
  }
}
```
